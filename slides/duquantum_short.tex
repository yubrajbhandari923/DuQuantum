\figframe[0.62]{The setup}{
  \item A storage resonator holds the qubit, about 4.4 photons of microwave light
  \item A pump and a leaky helper resonator keep it in shape; we simulated both together
}{p01}

\figframe[0.70]{The qubit is a ball in one of two valleys}{
  \item One valley means 0, the other means 1, and the pump keeps the ball where it is
  \item Switched on from empty, the pump builds the qubit in about 1 $\mu$s
}{p05}

\begin{frame}{Bit flips are a million times rarer than phase flips}
  \vfill
  \begin{columns}[c]
    \column{0.30\textwidth}\centering

    \column{0.30\textwidth}\centering
    {\Huge\color{tabblue} 1.94 $\mu$s}\\[0.5em] time to a phase flip
    \column{0.30\textwidth}\centering
    {\Huge\color{tabblue} $10^{6}$}\\[0.5em] how much rarer bit flips are
  \end{columns}
  \vfill
  \begin{itemize}\gap
    \item A bigger cat makes the valleys deeper, so the ball almost never hops
    \item Only one kind of error is left to correct, which makes error correction cheaper
  \end{itemize}
  \vfill
\end{frame}
