\begin{frame}{The problem: quantum computers make mistakes}
  \vfill
  \begin{itemize}\gap
    \item A qubit can be 0, 1, or both at once, but tiny disturbances from the outside world keep spoiling it
    \item It goes wrong in two ways: 0 and 1 swap (a \textbf{bit flip}), or the ``both at once'' gets scrambled (a \textbf{phase flip}). Fixing both kinds takes many extra qubits
  \end{itemize}
  \vfill
\end{frame}

\figframe{The idea: a cat qubit}{
  \item Store 0 and 1 as a ball resting in one of two valleys, and keep digging the valleys
  \item Deep valleys mean the ball almost never hops, so bit flips nearly vanish and only phase flips are left to fix
}{p05}{The two valleys, 0 and 1}

\figframe[0.62]{The device we simulated}{
  \item A tiny box (the memory) traps microwave light, about 4.4 particles of light (photons). This light is the qubit
  \item A pump and a leaky helper box add and remove light in pairs, which keeps the two valleys in place
}{p01}{Memory box, pump, helper box and its output line}

  \begin{itemize}\gap

  \end{itemize}
\end{frame}

\figframe[0.66]{Step 1: the pump turns empty light into a cat}{
  \item Snapshots of the light in the box: one blob splits into two, the cat that is 0 and 1 at once
  \item It takes about 1 microsecond, a millionth of a second
}{task1_1_wigner}{The light in the memory box, from 0 to 3 microseconds}

\figframe[0.66]{Step 2: bigger cats flip much less}{
  \item Each dot is one simulation. Each step down the scale means 10 times fewer bit flips
  \item For small cats, every extra photon makes bit flips about 20 times rarer
}{task2_1_gamma_x}{Bit flips per microsecond against cat size}

\figframe[0.66]{Phase flips: the stripes fade in about 2 microseconds}{
  \item The stripes between the blobs show the light is in both valleys at once. Each photon that leaks out of the box weakens them
  \item We timed the fading: a phase flip every 1.94 microseconds on average, as the textbook formula predicts
}{task1_3a_wigner}{The light in the memory box from 0 to 10 microseconds: the blobs stay, the stripes fade}

\begin{frame}{The big result: bit flips are a million times rarer}
  \vfill
  \begin{columns}[c]
    \column{0.30\textwidth}\centering
    {\Huge\color{tabblue} 2.6 s}\\[0.5em] average time to a bit flip
    \column{0.30\textwidth}\centering
    {\Huge\color{tabblue} 1.94 $\mu$s}\\[0.5em] average time to a phase flip
    \column{0.30\textwidth}\centering
    {\Huge\color{tabblue} 1{,}000{,}000$\times$}\\[0.5em] how much rarer bit flips are
  \end{columns}
  \vfill
  \begin{itemize}\gap
    \item At our working size, a cat of 4.4 photons
    \item If phase flips happened once a second, bit flips would happen about once every two weeks
  \end{itemize}
  \vfill
\end{frame}

\figframe[0.66]{Step 3: reading the qubit has a sweet spot}{
  \item A gentle push makes the helper box's signal go up for 0 and down for 1. Read too briefly and the signal is too faint; read too long and the cat may flip while we look
  \item Best balance: 0.4\% wrong answers. Our trick, making the cat bigger while reading, cuts the mistakes 13 to 18 times
}{task2_4_fom}{Readout mistakes against reading time, for four push strengths}

\begin{frame}{What we found}
  \vfill
  \begin{itemize}\gap
    \item The cat qubit keeps its bit for seconds: bit flips are a million times rarer than phase flips, so correcting errors gets much cheaper
    \item It can be read with 0.4\% wrong answers, and 13 to 18 times fewer when the cat is made bigger during the reading
  \end{itemize}
